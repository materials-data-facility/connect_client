# MDF Agent — 90-Second Demo

```bash
# First run — welcome panel
mdf

# Check setup
mdf doctor

# Publish a directory in one shot
mdf publish ./cobalt_oxide_dft/ \
  --title "DFT Formation Energies of Cobalt Oxide Polymorphs" \
  --author "Ben Blaiszik" \
  --submit

# Watch it go through curation
mdf watch

# Check the result
mdf status

# Grab a citation
mdf cite cobalt_oxide_dft_v1 -f bibtex --copy

# Open in browser
mdf open cobalt_oxide_dft_v1

# Search for it
mdf search "cobalt oxide"

# Preview the files
mdf preview cobalt_oxide_dft_v1

# Everything works as JSON for scripting
mdf status cobalt_oxide_dft_v1 --json | jq .submission.status
```
